const MAX = 1000, TTL = 86400, KVMAX = 20 * 1024 * 1024

async function kg(k) {
  return await storeGet(k)
}
async function kp(k, v, t) {
  await storeSet(k, v, t || TTL)
}
async function kd(k) {
  await storeDel(k)
}
async function gh(i) {
  return (await kg('c:' + i + ':h')) || []
}
async function sh(i, m) {
  let arr = m.slice(-MAX)
  while (arr.length > 1 && JSON.stringify(arr).length > KVMAX) arr = arr.slice(1)
  try {
    await kp('c:' + i + ':h', arr, TTL)
  } catch {    try { await kp('c:' + i + ':h', arr.slice(-8), TTL) } catch {}
  }
}
async function ah(i, r, c) {
  const h = await gh(i)
  h.push({ role: r, content: c })
  await sh(i, h)
  return h
}

async function tgS(id, t, op) {
  const bd = { chat_id: id, text: t, parse_mode: 'Markdown', disable_web_page_preview: true, ...op }
  const j = await (await fetch((await tgA()) + '/sendMessage', {

  delete bd.parse_mode
  return (await fetch((await tgA()) + '/sendMessage', {    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(bd)
  })).json()
}

async function tgE(id, mid, t, op) {
  const bd = { chat_id: id, message_id: mid, text: t, parse_mode: 'Markdown', disable_web_page_preview: true, ...(op || {}) }

async function tgDL(fp) {
  const r = await fetch('https://api.telegram.org/file/bot' + (await tgT()) + '/' + fp)
  const b = await r.arrayBuffer()
  const e = (fp.split('.').pop() || 'bin').toLowerCase()
  const mt = { jpg:'image/jpeg',jpeg:'image/jpeg',png:'image/png',gif:'image/gif',webp:'image/webp',pdf:'application/pdf',txt:'text/plain' }[e] || 'application/octet-stream'
  let s = ''
  new Uint8Array(b).forEach(c => { s += String.fromCharCode(c) })
  return { b64: btoa(s), mime: mt, ext: e, name: fp.split('/').pop() || 'file', size: b.byteLength }
}

const TEXTEXT = ['txt','md','markdown','csv','tsv','json','jsonl','ndjson','xml','yaml','yml','html','htm','css','scss','less','js','mjs','cjs','ts','tsx','jsx','py','java','c','cc','cpp','h','hpp','cs','go','rs','rb','php','sql','sh','bash','zsh','bat','ps1','ini','conf','cfg','toml','log','env','tex','r','kt','kts','swift','dart','vue','svelte','pl','lua','scala','gradle','properties','srt','vtt','gitignore','dockerfile','makefile','asm','m','vb','fs','clj','ex','exs','erl','hs','jl','nim','zig','proto','graphql','gql','rst','org','tsv']
function b64ToBytes(b64) { const bin = atob(b64); const u = new Uint8Array(bin.length); for (let i = 0; i < bin.length; i++) u[i] = bin.charCodeAt(i); return u }
function b64ToText(b64) { try { return new TextDecoder('utf-8', { fatal: false }).decode(b64ToBytes(b64)) } catch { return '' } }
function isTextFile(md) {
  if (!md) return false
  if (md.mime && (md.mime.indexOf('text/') === 0 || md.mime === 'application/json' || md.mime === 'application/xml' || md.mime === 'application/javascript')) return true
  const n = (md.name || '').toLowerCase()
  if (TEXTEXT.indexOf(md.ext) >= 0) return true
  if (n === 'dockerfile' || n === 'makefile' || n === '.gitignore' || n === '.env') return true
  return false
}

const UA = 'claude-cli/1.0.0 (external, cli)'
function hdrOA(k) { return { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + k, 'User-Agent': UA } }
function hdrANT(k) { return { 'Content-Type': 'application/json', 'x-api-key': k, 'anthropic-version': '2023-06-01', 'User-Agent': UA } }

async function apim() {
  const t = await at(), b = await base(), k = await akey()
  let ms = null, errs = []
  async function tryList(h, label) {
    if (ms) return
    try {
      const r = await fetch(b + '/v1/models', { headers: h })
      if (r.ok) {
        const d = await r.json()
        const arr = d.data || d.models || (Array.isArray(d) ? d : null)
        if (arr && arr.length) { ms = arr; return }
      }
      errs.push(label + ': ' + r.status + ' ' + (await r.text()).replace(/\s+/g, ' ').substring(0, 120))
    } catch (e) { errs.push(label + ': ' + e.message.substring(0, 80)) }
  }
  if (t === 'openai') { await tryList(hdrOA(k), 'OpenAI'); await tryList(hdrANT(k), 'Anthropic') }
  else { await tryList(hdrANT(k), 'Anthropic'); await tryList(hdrOA(k), 'OpenAI') }
  if (!ms) throw new Error('دریافت لیست مدل‌ها ممکن نشد\n' + errs.join('\n'))
  return ms.filter(m => m && (m.id || typeof m === 'string')).map(m => typeof m === 'string' ? { id: m } : m)
}

function convContent(c, to) {
  if (typeof c === 'string' || !Array.isArray(c)) return c
  return c.map(p => {
    if (!p || typeof p !== 'object') return p
    if (to === 'anthropic') {
      if (p.type === 'image_url' && p.image_url && p.image_url.url) {
        const mm = /^data:([^;]+);base64,(.+)$/.exec(p.image_url.url)
        if (mm) return { type: 'image', source: { type: 'base64', media_type: mm[1], data: mm[2] } }
      }      if (p.type === 'file' && p.file && p.file.file_data) {
        const fm = /^data:([^;]+);base64,(.+)$/.exec(p.file.file_data)
        if (fm) return { type: 'document', source: { type: 'base64', media_type: fm[1], data: fm[2] } }
      }
      if (p.type === 'document') return { type: 'document', source: p.source }
      return p
    }
    if (p.type === 'image' && p.source && p.source.type === 'base64') return { type: 'image_url', image_url: { url: 'data:' + p.source.media_type + ';base64,' + p.source.data, detail: 'high' } }
    if (p.type === 'document' && p.source && p.source.type === 'base64') return { type: 'file', file: { filename: p._name || 'file.pdf', file_data: 'data:' + p.source.media_type + ';base64,' + p.source.data } }
    return p
  })
}
function toAnthMsgs(m) { return m.filter(x => x.role !== 'system').map(x => ({ role: x.role, content: convContent(x.content, 'anthropic') })) }
function toOpenAIMsgs(sp, m) {
  const out = [{ role: 'system', content: sp }]
  for (const msg of m) { if (msg.role === 'system') continue; out.push({ role: msg.role, content: convContent(msg.content, 'openai') }) }
  return out
}
function parseNonStream(raw, type) {
  try {
    const o = JSON.parse(raw)
    if (type === 'openai') {
      const c = o.choices && o.choices[0] && (o.choices[0].message || o.choices[0].delta || {}).content
      if (typeof c === 'string') return c
      if (Array.isArray(c)) return c.map(x => x.text || x.content || '').join('')
    } else {
      if (Array.isArray(o.content)) return o.content.map(x => x.text || '').join('')
      if (typeof o.content === 'string') return o.content
    }
  } catch {}
  return ''
}

const DEF_OA = 'gpt-4o', DEF_ANT = 'claude-sonnet-4-6'
function isAnthModel(n) { return /claude/i.test(n || '') }
function isOAModel(n) { return /^(gpt|o[13]|chatgpt|gemini|deepseek|llama|mistral|qwen|grok)/i.test(n || '') }
function modelFor(fmt, model) {
  if (!model) return fmt === 'anthropic' ? DEF_ANT : DEF_OA
  if (fmt === 'anthropic' && isOAModel(model)) return DEF_ANT
  if (fmt === 'openai' && isAnthModel(model)) return model
  return model
}

async function aistream(s, m, model, oc, mx) {
  const t = await at(), k = await akey(), b = await base()
  let r = ''
  const mOA = modelFor('openai', model), mANT = modelFor('anthropic', model)
  if (t === 'openai') {
    try { r = await oaiStr(b, k, toOpenAIMsgs(s, m), mOA, oc, mx) } catch { r = await antStr(b, k, s, toAnthMsgs(m), mANT, oc, mx) }
  } else {    try { r = await antStr(b, k, s, toAnthMsgs(m), mANT, oc, mx) } catch { r = await oaiStr(b, k, toOpenAIMsgs(s, m), mOA, oc, mx) }
  }
  if (!r || !r.trim()) {
    if (t === 'openai') { try { r = await antStr(b, k, s, toAnthMsgs(m), mANT, oc, mx) } catch {} }
    else { try { r = await oaiStr(b, k, toOpenAIMsgs(s, m), mOA, oc, mx) } catch {} }
  }
  return r
}

async function oaiStr(b, k, m, model, oc, mx) {
  const r = await fetch(b + '/v1/chat/completions', {
    method: 'POST', headers: hdrOA(k),
    body: JSON.stringify({ model, messages: m, stream: true, max_tokens: mx || 4096 })
  })
  if (!r.ok) throw new Error('OpenAI Error (' + r.status + '): ' + (await r.text()).substring(0, 300))
  const rd = r.body.getReader(), dc = new TextDecoder()
  let buf = '', full = '', raw = '', any = false
  while (true) {
    const { done, value } = await rd.read()
    if (done) break
    const chunk = dc.decode(value, { stream: true })
    raw += chunk; buf += chunk
    const ls = buf.split('\n'); buf = ls.pop() || ''
    for (const l of ls) {
      const t = l.trim()
      if (!t.startsWith('data:')) continue
      const j = t.slice(6).trim()
      if (j === '[DONE]') continue
      try {
        const e = JSON.parse(j), c = e.choices?.[0]?.delta?.content || ''
        if (c) { full += c; any = true; oc(full) }
      } catch {}
    }
  }
  if (!any) { const p = parseNonStream(raw, 'openai'); if (p) { full = p; oc(full) } }
  return full
}

async function antStr(b, k, s, m, model, oc, mx) {
  const r = await fetch(b + '/v1/messages', {
    method: 'POST', headers: hdrANT(k),
    body: JSON.stringify({ model, max_tokens: mx || 4096, system: s, messages: m, stream: true })
  })
  if (!r.ok) throw new Error('Anthropic Error (' + r.status + '): ' + (await r.text()).substring(0, 300))
  const rd = r.body.getReader(), dc = new TextDecoder()
  let buf = '', full = '', raw = '', any = false
  while (true) {
    const { done, value } = await rd.read()
    if (done) break
    const chunk = dc.decode(value, { stream: true })    raw += chunk; buf += chunk
    const ls = buf.split('\n'); buf = ls.pop() || ''
    for (const l of ls) {
      const t = l.trim()
      if (!t.startsWith('data:')) continue
      const j = t.slice(6).trim()
      if (j === '[DONE]') continue
      try {
        const e = JSON.parse(j)
        if (e.type === 'content_block_delta' && e.delta?.text) { full += e.delta.text; any = true; oc(full) }
      } catch {}
    }
  }
  if (!any) { const p = parseNonStream(raw, 'anthropic'); if (p) { full = p; oc(full) } }
  return full
}
